{{- /* Accepts a dict with page and an optional pager. Only the HTML head supplies the current pager; sitemap entries and language alternates use base page URLs without initializing pagination as a side effect. */ -}} {{- $url := .page.Permalink -}} {{- if and .pager (gt .pager.PageNumber 1) -}} {{- $url = .pager.URL | absURL -}} {{- else if .page.IsHome -}} {{- $url = strings.TrimSuffix "index.html" $url -}} {{- end -}} {{- return $url -}}
