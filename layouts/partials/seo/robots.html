{{- /* Returns the robots meta value for a page, or "" when search engines may index it. The head partial and the sitemap both use this, so a page that asks search engines to skip it is never listed in the sitemap. - robots front matter wins (for example "noindex, follow"). - noindex: true keeps its original meaning, "noindex, nofollow". - Tag, category, and section list pages with fewer than 2 visible posts (archived posts are hidden) are thin pages that Google reports as soft 404s, so they get "noindex, follow". - The /blog/ section list duplicates /articles.html. CloudFront redirects it; this keeps it out of the index and the sitemap as well. */ -}} {{- $robots := "" -}} {{- with .Params.robots -}} {{- $robots = . -}} {{- else -}} {{- if .Params.noindex -}} {{- $robots = "noindex, nofollow" -}} {{- else if in (slice "term" "section") .Kind -}} {{- $visible := where .Pages "Params.maintenance_status" "ne" "archived" -}} {{- if or (lt (len $visible) 2) (and (eq .Kind "section") (eq .Section "blog")) -}} {{- $robots = "noindex, follow" -}} {{- end -}} {{- end -}} {{- end -}} {{- return $robots -}}
